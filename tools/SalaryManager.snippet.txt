function SalaryManager(){
 const [employees,setEmployees]=useState<any[]|null>(null),[periods,setPeriods]=useState<any[]|null>(null),[summary,setSummary]=useState<any|null>(null),[accounts,setAccounts]=useState<any[]>([]),[error,setError]=useState(''),[open,setOpen]=useState(false),[pay,setPay]=useState<any|null>(null),[edit,setEdit]=useState<any|null>(null); const now=new Date(); const [month,setMonth]=useState(`${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}`);
 const fm=(v:any)=>`$ ${Number(v||0).toLocaleString('es-AR',{maximumFractionDigits:2})}`; const fd=(v:any)=>v?new Date(`${String(v).slice(0,10)}T12:00:00`).toLocaleDateString('es-AR'):'—';
 const load=async()=>{try{const [e,p,s,a]=await Promise.all([api.get<any[]>('/api/salaries/employees'),api.get<any[]>('/api/salaries/periods?months_back=12&months_forward=12'),api.get<any>('/api/salaries/summary'),api.list<any>('accounts','?limit=500')]);setEmployees(e);setPeriods(p);setSummary(s);setAccounts(a.filter((x:any)=>x.is_active!==false));setError('')}catch(e:any){setError(e.message||String(e))}}; useEffect(()=>{void load()},[]); if(error)return <ErrorBox message={error} onRetry={load}/>; if(!employees||!periods||!summary)return <Loading/>;
 const rows=periods.filter((x:any)=>String(x.period_month||'').slice(0,7)===month); const total=rows.reduce((a:number,x:any)=>a+Number(x.base_amount||0)+Number(x.adjustments||0),0),paid=rows.reduce((a:number,x:any)=>a+Number(x.paid_amount||0),0),pending=rows.reduce((a:number,x:any)=>a+Number(x.balance||0),0); const ml=(()=>{const [y,m]=month.split('-').map(Number);const s=new Intl.DateTimeFormat('es-AR',{month:'long',year:'numeric'}).format(new Date(y,m-1,1));return s.charAt(0).toUpperCase()+s.slice(1)})();
 return <><div className="kpi-grid four"><Kpi label={`Sueldos - ${ml}`} value={fm(total)}/><Kpi label="Pagado" value={fm(paid)} tone="good"/><Kpi label="Pendiente" value={fm(pending)} tone={pending>0?'warn':undefined}/><Kpi label="Vencido acumulado" value={fm(summary.overdue)} tone={Number(summary.overdue)>0?'bad':'good'}/></div><Card><SectionTitle title="Liquidación mensual" subtitle="Lo que no se paga queda pendiente como obligación salarial." action={<label style={{display:'grid',gap:5,fontSize:12,fontWeight:700}}>Mes<input type="month" value={month} onChange={e=>setMonth(e.target.value)}/></label>}/>{rows.length===0?<div className="empty-state">No hay sueldos para {ml}.</div>:<div className="table-wrap"><table><thead><tr><th>Persona</th><th>Puesto</th><th>Vence</th><th>Base</th><th>Ajustes</th><th>Total</th><th>Pagado</th><th>Saldo</th><th>Estado</th><th>Acciones</th></tr></thead><tbody>{rows.map((r:any)=><tr key={r.id}><td><b>{r.employee_name}</b></td><td>{r.employee_role||'—'}</td><td>{fd(r.due_date)}</td><td>{fm(r.base_amount)}</td><td>{fm(r.adjustments)}</td><td><b>{fm(Number(r.base_amount||0)+Number(r.adjustments||0))}</b></td><td>{fm(r.paid_amount)}</td><td><b>{fm(r.balance)}</b></td><td><Status tone={r.status==='pagado'?'green':r.status==='vencido'?'red':r.status==='parcial'?'blue':'yellow'}>{r.status}</Status></td><td><div className="row-actions">{Number(r.balance)>0&&<button className="mini-button" onClick={()=>setPay(r)}>Pagar</button>}<button className="mini-button" onClick={()=>setEdit(r)}>Ajustar</button></div></td></tr>)}</tbody></table></div>}</Card><Card><SectionTitle title="Personal y sueldo base" action={<button className="primary-button" onClick={()=>setOpen(true)}>+ Persona</button>}/><div className="table-wrap"><table><thead><tr><th>Persona</th><th>Puesto</th><th>Sueldo mensual</th><th>Día pago</th><th>Desde</th><th>Hasta</th></tr></thead><tbody>{employees.map((e:any)=><tr key={e.id}><td><b>{e.name}</b></td><td>{e.role||'—'}</td><td>{fm(e.monthly_salary)}</td><td>{e.due_day}</td><td>{fd(e.start_date)}</td><td>{fd(e.end_date)}</td></tr>)}</tbody></table></div></Card>{open&&<SalaryEmployeeModal close={()=>setOpen(false)} done={async()=>{setOpen(false);await load()}}/>}{pay&&<SalaryPayModal row={pay} accounts={accounts} close={()=>setPay(null)} done={async()=>{setPay(null);await load()}}/>}{edit&&<SalaryAdjustModal row={edit} close={()=>setEdit(null)} done={async()=>{setEdit(null);await load()}}/>}</>
}
function SalaryEmployeeModal({close,done}:any){const [f,setF]=useState<any>({name:'',role:'',monthly_salary:'',due_day:5,start_date:new Date().toISOString().slice(0,10),end_date:''}),[saving,setSaving]=useState(false);const submit=async()=>{setSaving(true);try{await api.post('/api/salaries/employees',{...f,monthly_salary:Number(f.monthly_salary),due_day:Number(f.due_day),end_date:f.end_date||null});await done()}catch(e:any){alert(e.message)}finally{setSaving(false)}};return <div className="modal-backdrop"><div className="modal"><div className="modal-head"><h2>Nueva persona</h2><button className="close-button" onClick={close}>×</button></div><div className="form-grid"><label className="field"><span>Nombre *</span><input value={f.name} onChange={e=>setF({...f,name:e.target.value})}/></label><label className="field"><span>Puesto</span><input value={f.role} onChange={e=>setF({...f,role:e.target.value})}/></label><label className="field"><span>Sueldo mensual *</span><input type="number" value={f.monthly_salary} onChange={e=>setF({...f,monthly_salary:e.target.value})}/></label><label className="field"><span>Día pago</span><input type="number" min="1" max="31" value={f.due_day} onChange={e=>setF({...f,due_day:e.target.value})}/></label><label className="field"><span>Desde</span><input type="date" value={f.start_date} onChange={e=>setF({...f,start_date:e.target.value})}/></label><label className="field"><span>Hasta</span><input type="date" value={f.end_date} onChange={e=>setF({...f,end_date:e.target.value})}/></label></div><div className="modal-actions"><button className="ghost-button" onClick={close}>Cancelar</button><button className="primary-button" disabled={saving} onClick={submit}>Guardar</button></div></div></div>}
function SalaryPayModal({row,accounts,close,done}:any){const [f,setF]=useState<any>({account_id:accounts[0]?.id||'',amount:String(row.balance||''),payment_date:new Date().toISOString().slice(0,10)}),[saving,setSaving]=useState(false);const submit=async()=>{setSaving(true);try{await api.post(`/api/salaries/periods/${row.id}/payments`,{...f,amount:Number(f.amount)});await done()}catch(e:any){alert(e.message)}finally{setSaving(false)}};return <div className="modal-backdrop"><div className="modal"><div className="modal-head"><h2>Pagar sueldo - {row.employee_name}</h2><button className="close-button" onClick={close}>×</button></div><div className="form-grid"><label className="field"><span>Cuenta *</span><select value={f.account_id} onChange={e=>setF({...f,account_id:e.target.value})}>{accounts.map((a:any)=><option key={a.id} value={a.id}>{a.name}</option>)}</select></label><label className="field"><span>Monto</span><input type="number" max={Number(row.balance||0)} value={f.amount} onChange={e=>setF({...f,amount:e.target.value})}/></label><label className="field"><span>Fecha</span><input type="date" value={f.payment_date} onChange={e=>setF({...f,payment_date:e.target.value})}/></label></div><div className="modal-actions"><button className="ghost-button" onClick={close}>Cancelar</button><button className="primary-button" disabled={saving} onClick={submit}>Registrar pago</button></div></div></div>}
function SalaryAdjustModal({row,close,done}:any){const [adjustments,setAdjustments]=useState(String(row.adjustments||0)),[notes,setNotes]=useState(row.notes||''),[saving,setSaving]=useState(false);const submit=async()=>{setSaving(true);try{await fetch(`${(process.env.NEXT_PUBLIC_API_URL||'https://dirac-admin.onrender.com').replace(/\/$/,'')}/api/salaries/periods/${row.id}`,{method:'PATCH',headers:{'Content-Type':'application/json',...(process.env.NEXT_PUBLIC_API_KEY?{'X-API-Key':process.env.NEXT_PUBLIC_API_KEY}:{})},body:JSON.stringify({adjustments:Number(adjustments||0),notes})}).then(async r=>{if(!r.ok)throw new Error((await r.json()).detail||`Error ${r.status}`)});await done()}catch(e:any){alert(e.message)}finally{setSaving(false)}};return <div className="modal-backdrop"><div className="modal"><div className="modal-head"><h2>Ajustar - {row.employee_name}</h2><button className="close-button" onClick={close}>×</button></div><p className="modal-note">Positivo: horas extra/premio. Negativo: descuento/anticipo.</p><div className="form-grid"><label className="field"><span>Ajuste</span><input type="number" value={adjustments} onChange={e=>setAdjustments(e.target.value)}/></label><label className="field"><span>Detalle</span><input value={notes} onChange={e=>setNotes(e.target.value)}/></label></div><div className="modal-actions"><button className="ghost-button" onClick={close}>Cancelar</button><button className="primary-button" disabled={saving} onClick={submit}>Guardar</button></div></div></div>}
